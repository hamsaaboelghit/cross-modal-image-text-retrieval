# The reference values match the existing Baseline configuration.
REFERENCE = {
    "learning_rate": 1e-4,
    "batch_size": 8,
    "output_dim": 512,
    "dropout": 0.1,
    "temperature": 0.07,
}


def build_experiments():
    """Create one-factor-at-a-time experiments."""

    experiments = [
        {
            "name": "reference",
            "changed_parameter": "reference",
            "changed_value": "default",
            **REFERENCE,
        }
    ]

    variations = [
        ("learning_rate", 1e-3),
        ("learning_rate", 1e-5),

        ("batch_size", 16),
        ("batch_size", 32),

        ("output_dim", 256),

        ("dropout", 0.2),
        ("dropout", 0.3),

        ("temperature", 0.05),
        ("temperature", 0.1),
    ]

    for parameter, value in variations:
        config = dict(REFERENCE)
        config[parameter] = value

        experiments.append({
            "name": f"{parameter}_{str(value).replace('.', '_').replace('-', 'm')}",
            "changed_parameter": parameter,
            "changed_value": value,
            **config,
        })

    return experiments


def validate_experiment_design(experiments):
    """Verify that each experiment changes one parameter only."""

    reference_values = {
        key: REFERENCE[key]
        for key in REFERENCE
    }

    names = [item["name"] for item in experiments]

    if len(names) != len(set(names)):
        raise ValueError("Experiment names must be unique.")

    if len(experiments) != 10:
        raise ValueError(
            "Expected one reference and nine variations."
        )

    for experiment in experiments:
        differences = [
            key
            for key in reference_values
            if experiment[key] != reference_values[key]
        ]

        if experiment["name"] == "reference":
            if differences:
                raise ValueError(
                    "Reference experiment must use all default values."
                )
        elif differences != [experiment["changed_parameter"]]:
            raise ValueError(
                f"{experiment['name']} changes unexpected parameters: "
                f"{differences}"
            )

    print("[PASS] One-factor-at-a-time experiment design.")
    print(f"[PASS] Total experiments: {len(experiments)}")


def build_run_metadata(experiments, epochs):
    """Return the exact settings required to safely resume a run."""
    return {
        "schema_version": RUN_SCHEMA_VERSION,
        "model_type": MODEL_TYPE,
        "fold": FOLD,
        "seed": SEED,
        "epochs": epochs,
        "selection_split": "validation",
        "experiments": [
            {key: experiment[key] for key in (
                "name",
                "changed_parameter",
                "changed_value",
                "learning_rate",
                "batch_size",
                "output_dim",
                "dropout",
                "temperature",
            )}
            for experiment in experiments
        ],
    }


def validate_resume_metadata(existing, expected):
    """Reject resume attempts that would mix incompatible runs."""
    if existing != expected:
        raise ValueError(
            "Existing hyperparameter run metadata does not match this "
            "command. Keep the original --epochs/--only settings, or use "
            "--overwrite to intentionally start a new run."
        )


def load_resume_rows(metadata, expected):
    metadata_path = OUTPUT_DIR / "hyperparameter_run_metadata.json"
    results_path = OUTPUT_DIR / "hyperparameter_results.json"

    if not metadata_path.is_file() or not results_path.is_file():
        raise FileNotFoundError(
            "Cannot resume: both hyperparameter_run_metadata.json and "
            "hyperparameter_results.json must exist."
        )

    with metadata_path.open("r", encoding="utf-8") as file:
        existing_metadata = json.load(file)

    validate_resume_metadata(existing_metadata, expected)

    with results_path.open("r", encoding="utf-8") as file:
        rows = json.load(file)

    if not isinstance(rows, list):
        raise ValueError("Existing hyperparameter results must be a JSON list.")

    expected_names = {item["name"] for item in metadata["experiments"]}
    unexpected_names = {
        row.get("experiment")
        for row in rows
        if row.get("experiment") not in expected_names
    }
    if unexpected_names:
        raise ValueError(
            "Existing results contain experiments outside this run: "
            f"{sorted(unexpected_names)}"
        )

    return rows


def atomic_write_json(path, payload):
    """Write JSON through a temporary sibling file to avoid partial results."""
    temporary_path = path.with_suffix(path.suffix + ".tmp")
    with temporary_path.open("w", encoding="utf-8") as file:
        json.dump(payload, file, indent=4, ensure_ascii=False)
        file.flush()
    temporary_path.replace(path)


def find_best_checkpoint(experiment_name):
    checkpoint_dir = (
        PROJECT_ROOT
        / "checkpoints"
        / "experiments"
        / experiment_name
        / "best"
    )

    checkpoint_path = tf.train.latest_checkpoint(
        str(checkpoint_dir)
    )

    if checkpoint_path is None:
        raise FileNotFoundError(
            f"No best checkpoint found in: {checkpoint_dir}"
        )

    return checkpoint_path


def evaluate_experiment(experiment):
    """Evaluate the best checkpoint on the validation split."""

    name = experiment["name"]
    output_dim = experiment["output_dim"]

    print("\n" + "-" * 70)
    print(f"VALIDATION RETRIEVAL EVALUATION: {name}")
    print("-" * 70)

    vocabulary = load_vocabulary(VOCAB_PATH)

    model = BaselineRetrievalModel(
        vocab_size=len(vocabulary),
        embedding_dim=300,
        output_dim=output_dim,
        dropout=experiment["dropout"],
    )

    # Build model variables before restoring the checkpoint.
    dummy_images = tf.zeros(
        (1, 224, 224, 3),
        dtype=tf.float32,
    )

    dummy_captions = tf.fill(
        (1, 5, 34),
        tf.constant(4, dtype=tf.int32),
    )

    model(
        dummy_images,
        dummy_captions,
        training=False,
    )

    checkpoint_path = find_best_checkpoint(name)

    checkpoint = tf.train.Checkpoint(model=model)
    status = checkpoint.restore(checkpoint_path)
    status.assert_existing_objects_matched()
    status.expect_partial()

    print("[PASS] Best model checkpoint restored.")

    validation_data = Flickr30kDataset(
        fold_number=FOLD,
        split="validation",
    )

    validation_dataset = validation_data.as_dataset(
        batch_size=EVAL_BATCH_SIZE,
        shuffle=False,
        drop_remainder=False,
        seed=SEED,
    )

    evaluator = RetrievalEvaluator(
        model=model,
        query_batch_size=32,
        candidate_chunk_size=8192,
        ks=(1, 5, 10),
    )

    results = evaluator.evaluate(validation_dataset)

    text_to_image = results["text_to_image"]
    image_to_text = results["image_to_text"]

    overall_mean_recall = (
        text_to_image["mean_recall"]
        + image_to_text["mean_recall"]
    ) / 2.0

    overall_mrr = (
        text_to_image["mrr"]
        + image_to_text["mrr"]
    ) / 2.0

    metrics = {
        "checkpoint_path": str(checkpoint_path),
        "validation_images": results["num_images"],
        "embedding_dim": results["embedding_dim"],

        "text_to_image_r1": text_to_image["recall@1"],
        "text_to_image_r5": text_to_image["recall@5"],
        "text_to_image_r10": text_to_image["recall@10"],
        "text_to_image_mean_recall": text_to_image["mean_recall"],
        "text_to_image_mrr": text_to_image["mrr"],
        "text_to_image_median_rank": text_to_image["median_rank"],

        "image_to_text_r1": image_to_text["recall@1"],
        "image_to_text_r5": image_to_text["recall@5"],
        "image_to_text_r10": image_to_text["recall@10"],
        "image_to_text_mean_recall": image_to_text["mean_recall"],
        "image_to_text_mrr": image_to_text["mrr"],
        "image_to_text_median_rank": image_to_text["median_rank"],

        "overall_mean_recall": overall_mean_recall,
        "overall_mrr": overall_mrr,
        "evaluation_split": "validation",
        "test_set_used": False,
    }

    print("\nValidation metrics:")
    print(f"Text -> Image R@1: {metrics['text_to_image_r1']:.4f}")
    print(f"Image -> Text R@1: {metrics['image_to_text_r1']:.4f}")
    print(f"Overall Mean Recall: {overall_mean_recall:.4f}")
    print(f"Overall MRR: {overall_mrr:.4f}")

    return metrics


def get_best_training_losses(history):
    # Keep the total objective for diagnostics, but report retrieval loss
    # separately because it is the shared objective used for validation
    # checkpoint selection across the project's model architectures.
    train_losses = history.get("train_loss", [])
    train_retrieval_losses = history.get(
        "train_retrieval_loss",
        train_losses,
    )
    val_retrieval_losses = history.get(
        "val_retrieval_loss",
        history.get("val_loss", []),
    )

    if not train_losses or not val_retrieval_losses:
        raise ValueError(
            "Training history does not contain train/validation losses."
        )

    if not (
        len(train_losses)
        == len(train_retrieval_losses)
        == len(val_retrieval_losses)
    ):
        raise ValueError(
            "Training objective, retrieval, and validation history "
            "lengths do not match."
        )

    best_epoch_index = int(np.argmin(val_retrieval_losses))

    return {
        "best_epoch": best_epoch_index + 1,
        "best_train_loss": float(train_losses[best_epoch_index]),
        "best_train_retrieval_loss": float(
            train_retrieval_losses[best_epoch_index]
        ),
        "best_val_loss": float(val_retrieval_losses[best_epoch_index]),
        "best_val_retrieval_loss": float(
            val_retrieval_losses[best_epoch_index]
        ),
        "final_train_loss": float(train_losses[-1]),
        "final_train_retrieval_loss": float(train_retrieval_losses[-1]),
        "final_val_loss": float(val_retrieval_losses[-1]),
        "final_val_retrieval_loss": float(val_retrieval_losses[-1]),
        "epochs_completed": len(train_losses),
    }


def build_result_row(experiment, history, metrics, duration, epochs_requested):
    row = {
        "experiment": experiment["name"],
        "status": "completed",
        "changed_parameter": experiment["changed_parameter"],
        "changed_value": experiment["changed_value"],

        "learning_rate": experiment["learning_rate"],
        "batch_size": experiment["batch_size"],
        "output_dim": experiment["output_dim"],
        "dropout": experiment["dropout"],
        "temperature": experiment["temperature"],

        "seed": SEED,
        "fold": FOLD,
        "optimizer": "AdamW",
        "weight_decay": 1e-4,
        "scheduler": "none",
        "epochs_requested": epochs_requested,
        "early_stopping_patience": 3,
        "training_time_minutes": duration,

        **get_best_training_losses(history),
        **metrics,
    }

    return row

def save_best_configuration(rows):
    successful = [
        row
        for row in rows
        if row.get("status") == "completed"
        and "overall_mean_recall" in row
    ]

    if not successful:
        print(
            "\n[WARNING] No completed experiment is available "
            "for best-configuration selection."
        )
        return None

    # Primary criterion: highest validation mean recall.
    # Tie-breaker: lowest best validation loss.
    best = max(
        successful,
        key=lambda row: (
            row["overall_mean_recall"],
            -row.get("best_val_retrieval_loss", row["best_val_loss"]),
        ),
    )

    report = {
        "selection_split": "validation",
        "test_set_used_for_selection": False,
        "selection_criterion": (
            "Highest overall validation mean recall; "
            "lowest best validation loss used as tie-breaker."
        ),
        "best_experiment": best["experiment"],
        "best_configuration": {
            "learning_rate": best["learning_rate"],
            "batch_size": best["batch_size"],
            "output_dim": best["output_dim"],
            "dropout": best["dropout"],
            "temperature": best["temperature"],
        },
        "best_validation_metrics": {
            "overall_mean_recall": best["overall_mean_recall"],
            "overall_mrr": best["overall_mrr"],
            "best_val_retrieval_loss": best.get(
                "best_val_retrieval_loss",
                best["best_val_loss"],
            ),
        },
    }

    path = OUTPUT_DIR / "best_hyperparameter_configuration.json"

    atomic_write_json(path, report)

    print("\n" + "=" * 70)
    print("BEST HYPERPARAMETER CONFIGURATION")
    print("=" * 70)
    print("Experiment:", best["experiment"])
    print("Configuration:", report["best_configuration"])
    print(
        "Validation Mean Recall:",
        f"{best['overall_mean_recall']:.6f}",
    )
    print(
        "Best Validation Loss:",
        f"{best['best_val_loss']:.6f}",
    )
    print("Selection used Validation only.")
    print("Test set was NOT used.")

    return report


def create_plots(rows):
    import matplotlib.pyplot as plt

    successful = [
        row
        for row in rows
        if row.get("status") == "completed"
        and "overall_mean_recall" in row
    ]

    if not successful:
        print("[WARNING] No successful experiments to plot.")
        return

    plot_dir = OUTPUT_DIR / "plots"
    plot_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

    parameter_labels = {
        "learning_rate": "Learning Rate",
        "batch_size": "Batch Size",
        "output_dim": "Retrieval Embedding Dimension",
        "dropout": "Dropout",
        "temperature": "InfoNCE Temperature",
    }

    for parameter, title in parameter_labels.items():
        selected = [
            row
            for row in successful
            if row["changed_parameter"] in {parameter, "reference"}
        ]

        if not selected:
            continue

        selected.sort(
            key=lambda row: (
                row[parameter],
                row["experiment"],
            )
        )

        labels = [
            str(row[parameter])
            for row in selected
        ]

        mean_recalls = [
            row["overall_mean_recall"]
            for row in selected
        ]

        val_losses = [
            row["best_val_loss"]
            for row in selected
        ]

        figure, axis = plt.subplots(figsize=(8, 5))
        axis.bar(labels, mean_recalls)
        axis.set_title(f"{title} vs Validation Mean Recall")
        axis.set_xlabel(parameter.replace("_", " ").title())
        axis.set_ylabel("Overall Validation Mean Recall")
        axis.set_ylim(0, 1)
        axis.grid(axis="y", alpha=0.25)
        figure.tight_layout()

        figure.savefig(
            plot_dir / f"{parameter}_mean_recall.png",
            dpi=200,
        )
        plt.close(figure)

        figure, axis = plt.subplots(figsize=(8, 5))
        axis.bar(labels, val_losses)
        axis.set_title(f"{title} vs Best Validation Loss")
        axis.set_xlabel(parameter.replace("_", " ").title())
        axis.set_ylabel("Best Validation Loss")
        axis.grid(axis="y", alpha=0.25)
        figure.tight_layout()

        figure.savefig(
            plot_dir / f"{parameter}_validation_loss.png",
            dpi=200,
        )
        plt.close(figure)

    # Overall experiment comparison.
    labels = [
        row["experiment"]
        for row in successful
    ]
    values = [
        row["overall_mean_recall"]
        for row in successful
    ]

    figure, axis = plt.subplots(
        figsize=(max(10, len(labels) * 1.2), 6)
    )
    axis.bar(labels, values)
    axis.set_title("Hyperparameter Experiments - Validation Mean Recall")
    axis.set_ylabel("Overall Validation Mean Recall")
    axis.set_ylim(0, 1)
    axis.tick_params(axis="x", rotation=45)
    axis.grid(axis="y", alpha=0.25)
    figure.tight_layout()

    figure.savefig(
        plot_dir / "all_experiments_mean_recall.png",
        dpi=200,
    )
    plt.close(figure)

    print("\nPlots saved in:", plot_dir)


def run_all(experiments, epochs=5, resume=False, overwrite=False):
    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    metadata = build_run_metadata(experiments, epochs)
    metadata_path = OUTPUT_DIR / "hyperparameter_run_metadata.json"
    results_path = OUTPUT_DIR / "hyperparameter_results.json"

    if resume:
        rows = load_resume_rows(metadata, metadata)
        print(
            f"[RESUME] Loaded {len(rows)} saved experiment records."
        )
    else:
        existing_outputs = metadata_path.exists() or results_path.exists()
        if existing_outputs and not overwrite:
            raise FileExistsError(
                "Previous hyperparameter results already exist. Use "
                "--resume to continue the same run, or --overwrite to "
                "start a fresh run and replace its result files."
            )
        rows = []
        atomic_write_json(metadata_path, metadata)

    completed_names = {
        row["experiment"]
        for row in rows
        if row.get("status") == "completed"
        and "overall_mean_recall" in row
    }

    for index, experiment in enumerate(experiments, start=1):
        if experiment["name"] in completed_names:
            print(
                f"[SKIP] {experiment['name']} already completed "
                "and has validation metrics."
            )
            continue

        # Replace an earlier failed/incomplete record if this experiment
        # is being retried during a resumed run.
        rows = [
            row for row in rows
            if row.get("experiment") != experiment["name"]
        ]
        print("\n\n")
        print("#" * 75)
        print(
            f"EXPERIMENT {index}/{len(experiments)}: "
            f"{experiment['name']}"
        )
        print("#" * 75)

        start_time = time.time()

        try:
            history = run_experiment(
                experiment_name=experiment["name"],
                model_type=MODEL_TYPE,
                epochs=epochs,
                batch_size=experiment["batch_size"],
                seed=SEED,
                fold=FOLD,
                learning_rate=experiment["learning_rate"],
                temperature=experiment["temperature"],
                output_dim=experiment["output_dim"],
                dropout=experiment["dropout"],
                log_dir="runs/hyperparameter_experiments",
            )

            duration = (time.time() - start_time) / 60.0

            metrics = evaluate_experiment(experiment)

            row = build_result_row(
                experiment=experiment,
                history=history,
                metrics=metrics,
                duration=duration,
                epochs_requested=epochs,
            )

            rows.append(row)

            print(
                f"\n[PASS] {experiment['name']} "
                "training and validation evaluation completed."
            )

        except tf.errors.ResourceExhaustedError as error:
            duration = (time.time() - start_time) / 60.0

            rows.append({
                "experiment": experiment["name"],
                "status": "failed_out_of_memory",
                "changed_parameter": experiment["changed_parameter"],
                "changed_value": experiment["changed_value"],
                "learning_rate": experiment["learning_rate"],
                "batch_size": experiment["batch_size"],
                "output_dim": experiment["output_dim"],
                "dropout": experiment["dropout"],
                "temperature": experiment["temperature"],
                "training_time_minutes": duration,
                "error": str(error)[:1000],
            })

            print(
                f"\n[OUT OF MEMORY] {experiment['name']} "
                "could not run with the available memory."
            )

        except Exception as error:
            duration = (time.time() - start_time) / 60.0

            rows.append({
                "experiment": experiment["name"],
                "status": "failed",
                "changed_parameter": experiment["changed_parameter"],
                "changed_value": experiment["changed_value"],
                "learning_rate": experiment["learning_rate"],
                "batch_size": experiment["batch_size"],
                "output_dim": experiment["output_dim"],
                "dropout": experiment["dropout"],
                "temperature": experiment["temperature"],
                "training_time_minutes": duration,
                "error": str(error)[:1000],
            })

            print(f"\n[FAILED] {experiment['name']}")
            print(error)
            traceback.print_exc()

        finally:
            tf.keras.backend.clear_session()
            gc.collect()

            # Save progress after every experiment.
            save_results(rows)

    save_best_configuration(rows)
    create_plots(rows)
    save_results(rows)

    return rows


def main():
    parser = argparse.ArgumentParser(
        description=(
            "Run one-factor-at-a-time hyperparameter experiments "
            "for the VisionMatch AI Baseline model."
        )
    )

    parser.add_argument(
        "--epochs",
        type=int,
        default=5,
        help="Maximum training epochs per experiment.",
    )

    parser.add_argument(
        "--only",
        nargs="+",
        default=None,
        help=(
            "Optional experiment names to run. "
            "By default, all experiments are executed."
        ),
    )

    parser.add_argument(
        "--list",
        action="store_true",
        help="List experiment configurations without training.",
    )

    parser.add_argument(
        "--resume",
        action="store_true",
        help=(
            "Resume the exact previous run, skipping completed experiments "
            "and retrying failed/incomplete ones."
        ),
    )

    parser.add_argument(
        "--overwrite",
        action="store_true",
        help=(
            "Start a fresh run and replace existing result files. "
            "Cannot be combined with --resume."
        ),
    )

    args = parser.parse_args()

    if args.epochs <= 0:
        parser.error("--epochs must be positive.")

    if args.resume and args.overwrite:
        parser.error("--resume and --overwrite cannot be used together.")

    experiments = build_experiments()
    validate_experiment_design(experiments)

    if args.only:
        requested = set(args.only)
        available = {
            experiment["name"]
            for experiment in experiments
        }

        unknown = requested - available

        if unknown:
            parser.error(
                f"Unknown experiment names: {sorted(unknown)}"
            )

        experiments = [
            experiment
            for experiment in experiments
            if experiment["name"] in requested
        ]

    print("\nEXPERIMENT PLAN")
    print("=" * 90)
    print("Model scope: BaselineRetrievalModel only")
    print("Selection: Validation overall mean recall; retrieval loss tie-break")
    print("Test split: not used for tuning or selection")

    for experiment in experiments:
        print(
            f"{experiment['name']:<25} "
            f"LR={experiment['learning_rate']:<8} "
            f"Batch={experiment['batch_size']:<3} "
            f"Dim={experiment['output_dim']:<3} "
            f"Dropout={experiment['dropout']:<4} "
            f"Temp={experiment['temperature']}"
        )

    if args.list:
        print("\nList-only mode. No training was started.")
        return

    print("\nWARNING:")
    print(
        "This command starts TensorFlow training for each "
        "selected experiment."
    )
    print(
        "Run it only when the GPU is available."
    )

    run_all(
        experiments=experiments,
        epochs=args.epochs,
        resume=args.resume,

"""Run the controlled Phase 5 data and initialization experiment matrix."""
import argparse

ROOT = Path(__file__).resolve().parents[2]
FOLD, SEED = 1, 42
LR, OUTPUT_DIM, DROPOUT = 1e-4, 512, 0.1
OUT = ROOT / "results" / "phase5_data_strategy"
VOCAB = ROOT / "data" / "processed" / "flickr30k" / "vocabularies" / "fold_1_vocab.json"


def build_experiments():
    base = {
        "model_type": "full", "encoder_type": "lstm",
        "use_cross_modal_attention": False, "unfreeze_last_block": False,
        "caption_count": 5, "image_augmentation": "none",
        "negative_sampling_strategy": "random", "pretrained": True,
        "train_all_backbone": False, "loss_type": "info_nce", "margin": None,
    }
    specs = [
        ("data_reference", {}),
        ("one_caption", {"caption_count": 1}),
        ("light_augmentation", {"image_augmentation": "light"}),
        ("triplet_random", {"loss_type": "triplet"}),
        ("triplet_hardest", {"loss_type": "triplet", "negative_sampling_strategy": "hardest"}),
        ("resnet50_pretrained_finetune", {"train_all_backbone": True}),
        ("resnet50_scratch_finetune", {"pretrained": False, "train_all_backbone": True}),
    ]
    return [{"name": name, **base, **changes} for name, changes in specs]


def metadata(all_experiments, epochs, batch_size, train_steps_per_epoch, validation_steps):
    return {
        "schema_version": 3, "fold": FOLD, "seed": SEED,
        "epochs": epochs, "batch_size": batch_size,
        "train_steps_per_epoch": train_steps_per_epoch,
        "validation_steps": validation_steps, "learning_rate": LR,
        "output_dim": OUTPUT_DIM, "dropout": DROPOUT,
        "selection_split": "validation", "test_set_used": False,
        "experiments": all_experiments,
    }


def atomic_json(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + ".tmp")
    with temp.open("w", encoding="utf-8") as file:
        json.dump(value, file, indent=2, ensure_ascii=False)
        file.flush()
    temp.replace(path)


def evaluate_checkpoint(config, batch_size):
    vocab = load_vocabulary(VOCAB)
    model = CrossModalRetrievalModel(
        vocab_size=len(vocab), encoder_type="lstm", embedding_dim=300,
        hidden_dim=512, output_dim=OUTPUT_DIM, num_layers=1,
        dropout=DROPOUT, bidirectional=False, use_attention=True,
        pretrained=config["pretrained"], unfreeze_last_block=False,
        train_all_backbone=config["train_all_backbone"],
        use_cross_modal_attention=False,
    )
    model(
        tf.zeros((1, 224, 224, 3), tf.float32),
        tf.fill((1, 5, 34), tf.constant(4, tf.int32)),
        training=False,
    )
    checkpoint_dir = ROOT / "checkpoints" / "experiments" / config["name"] / "best"
    checkpoint = tf.train.latest_checkpoint(str(checkpoint_dir))
    if checkpoint is None:
        raise FileNotFoundError(f"No best checkpoint in {checkpoint_dir}")
    status = tf.train.Checkpoint(model=model).restore(checkpoint)
    status.assert_existing_objects_matched()
    status.expect_partial()

    val = Flickr30kDataset(FOLD, "validation")
    val_ds = val.as_dataset(
        batch_size=batch_size, shuffle=False, drop_remainder=False, seed=SEED
    )
    metrics = RetrievalEvaluator(
        model, query_batch_size=32, candidate_chunk_size=8192, ks=(1, 5, 10)
    ).evaluate(val_ds)
    metrics["checkpoint_path"] = checkpoint
    return metrics


def save_results(rows):
    OUT.mkdir(parents=True, exist_ok=True)
    atomic_json(OUT / "phase5_results.json", rows)
    fields = [
        "experiment", "status", "batch_size", "epochs_requested",
        "train_steps_per_epoch", "validation_steps",
        "validation_loss_is_partial", "loss_type", "caption_count",
        "image_augmentation", "negative_sampling_strategy", "pretrained",
        "train_all_backbone", "epochs_completed", "best_val_retrieval_loss",
        "final_val_retrieval_loss", "training_time_minutes",
        "overall_validation_mean_recall", "text_to_image_r1",
        "text_to_image_r5", "text_to_image_r10", "image_to_text_r1",
        "image_to_text_r5", "image_to_text_r10", "text_to_image_mrr",
        "image_to_text_mrr", "checkpoint_path", "error",
    ]
    path, temp = OUT / "phase5_comparison.csv", OUT / "phase5_comparison.csv.tmp"
    with temp.open("w", newline="", encoding="utf-8-sig") as file:
        writer = csv.DictWriter(file, fieldnames=fields, extrasaction="ignore")
        writer.writeheader()
        writer.writerows(rows)
        file.flush()
    temp.replace(path)


def run_all(
    experiments, all_experiments, epochs, batch_size,
    train_steps_per_epoch, validation_steps, resume, overwrite
):
    OUT.mkdir(parents=True, exist_ok=True)
    meta_path, results_path = OUT / "phase5_run_metadata.json", OUT / "phase5_results.json"
    expected = metadata(
        all_experiments, epochs, batch_size,
        train_steps_per_epoch, validation_steps,
    )
    if resume:
        if not meta_path.is_file() or not results_path.is_file():
            raise FileNotFoundError("Resume needs existing Phase 5 metadata and results.")
        if json.loads(meta_path.read_text(encoding="utf-8")) != expected:
            raise ValueError("Saved Phase 5 settings differ; use matching settings or --overwrite.")
        rows = json.loads(results_path.read_text(encoding="utf-8"))
    else:
        if (meta_path.exists() or results_path.exists()) and not overwrite:
            raise FileExistsError("Phase 5 results exist; use --resume or --overwrite.")
        rows = []
        atomic_json(meta_path, expected)

    completed = {
        r["experiment"] for r in rows
        if r.get("status") == "completed" and "overall_validation_mean_recall" in r
    }
    for i, config in enumerate(experiments, 1):
        name = config["name"]
        if name in completed:
            print(f"[SKIP] {name}")
            continue
        rows = [r for r in rows if r.get("experiment") != name]
        started = time.time()
        print(f"\nPHASE 5 {i}/{len(experiments)}: {name}")
        try:
            history = run_experiment(
                experiment_name=name, model_type=config["model_type"],
                encoder_type=config["encoder_type"],
                use_cross_modal_attention=config["use_cross_modal_attention"],
                pretrained=config["pretrained"],
                unfreeze_last_block=config["unfreeze_last_block"],
                train_all_backbone=config["train_all_backbone"],
                caption_count=config["caption_count"],
                image_augmentation=config["image_augmentation"],
                negative_sampling_strategy=config["negative_sampling_strategy"],
                epochs=epochs, batch_size=batch_size, seed=SEED, fold=FOLD,
                learning_rate=LR, loss_type=config["loss_type"],
                margin=config["margin"], output_dim=OUTPUT_DIM,
                dropout=DROPOUT, log_dir="runs/phase5_data_strategy",
                max_train_steps_per_epoch=train_steps_per_epoch,
                max_validation_steps=validation_steps,
            )
            metrics = evaluate_checkpoint(config, batch_size)
            t2i, i2t = metrics["text_to_image"], metrics["image_to_text"]
            val_losses = history.get("val_retrieval_loss") or history["val_loss"]
            row = {
                "experiment": name, "status": "completed",
                "batch_size": batch_size, "epochs_requested": epochs,
                "train_steps_per_epoch": train_steps_per_epoch,
                "validation_steps": validation_steps,
                "validation_loss_is_partial": True,
                "loss_type": config["loss_type"],
                "caption_count": config["caption_count"],
                "image_augmentation": config["image_augmentation"],
                "negative_sampling_strategy": config["negative_sampling_strategy"],
                "pretrained": config["pretrained"],
                "train_all_backbone": config["train_all_backbone"],
                "epochs_completed": len(history["train_loss"]),
                "best_val_retrieval_loss": min(val_losses),
                "final_val_retrieval_loss": val_losses[-1],
                "training_time_minutes": (time.time() - started) / 60.0,
                "overall_validation_mean_recall": (t2i["mean_recall"] + i2t["mean_recall"]) / 2.0,
                "text_to_image_r1": t2i["recall@1"],
                "text_to_image_r5": t2i["recall@5"],
                "text_to_image_r10": t2i["recall@10"],
                "image_to_text_r1": i2t["recall@1"],
                "image_to_text_r5": i2t["recall@5"],
                "image_to_text_r10": i2t["recall@10"],
                "text_to_image_mrr": t2i["mrr"],
                "image_to_text_mrr": i2t["mrr"],
                "checkpoint_path": metrics["checkpoint_path"],
                "evaluation_split": "validation", "test_set_used": False,
            }
            rows.append(row)
            print(f"[PASS] {name}: validation Mean Recall={row['overall_validation_mean_recall']:.4f}")
        except Exception as error:
            rows.append({
                "experiment": name, "status": "failed",
                "training_time_minutes": (time.time() - started) / 60.0,
                "error": str(error)[:1500],
            })
            print(f"[FAILED] {name}: {error}")
            traceback.print_exc()
        finally:
            tf.keras.backend.clear_session()
            save_results(rows)
    save_results(rows)
    return rows


def main():
    parser = argparse.ArgumentParser(description="Phase 5 controlled data and initialization experiments.")
    parser.add_argument("--epochs", type=int, default=5)
    parser.add_argument(
        "--train-steps-per-epoch", type=int, default=250,
        help="Training batches per epoch (default: 250).",
    )
    parser.add_argument(
        "--validation-steps", type=int, default=100,
        help="Validation batches per epoch; final retrieval evaluation still uses the full validation split.",
    )
    parser.add_argument(
        "--batch-size", type=int, default=4,
        help="Per-device batch size. Default 4 is conservative for GPUs with 6 GB VRAM.",
    )
    parser.add_argument("--list", action="store_true")
    parser.add_argument("--only", nargs="+")
    parser.add_argument("--resume", action="store_true")
    parser.add_argument("--overwrite", action="store_true")
    args = parser.parse_args()
    if args.epochs <= 0:
        parser.error("--epochs must be positive.")
    if args.batch_size <= 0:
        parser.error("--batch-size must be positive.")
    if args.train_steps_per_epoch <= 0:
        parser.error("--train-steps-per-epoch must be positive.")
    if args.validation_steps <= 0:
        parser.error("--validation-steps must be positive.")
    if args.resume and args.overwrite:
        parser.error("--resume and --overwrite cannot be combined.")

    all_experiments = build_experiments()
    experiments = all_experiments
    if args.only:
        requested, known = set(args.only), {x["name"] for x in experiments}
        if requested - known:
            parser.error(f"Unknown experiments: {sorted(requested - known)}")
        experiments = [x for x in experiments if x["name"] in requested]

    print("PHASE 5 EXPERIMENT PLAN | ResNet50 + LSTM | Fold 1 | Seed 42")
    print("Validation uses all five captions. Test split is never loaded.")
    for x in experiments:
        print(
            f"{x['name']:<34} captions={x['caption_count']} "
            f"augmentation={x['image_augmentation']} loss={x['loss_type']} "
            f"negative={x['negative_sampling_strategy']} "
            f"pretrained={x['pretrained']} train_backbone={x['train_all_backbone']}"
        )
    if args.list:
        print("List-only mode. No training was started.")
        return
    run_all(
        experiments, all_experiments, args.epochs, args.batch_size,
        args.train_steps_per_epoch, args.validation_steps,
        args.resume, args.overwrite,
    )


if __name__ == "__main__":
    main()
